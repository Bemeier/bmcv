#!args --rate=100ms --emit=ui
# Holding a ctrl button past UI_T_HOLD (100ms) latches its shift mode; the
# mode's own button taps out of it again. QNT is the exception that keeps its
# buttons as a semitone keyboard, so other ctrl taps do not exit it.
0ms     slider 0.0
100ms   ctrl QNT 1
400ms   ctrl QNT 0
600ms   ctrl STA 1
700ms   ctrl STA 0
900ms   ctrl QNT 1
920ms   ctrl QNT 0
1200ms  ctrl MON 1
1500ms  ctrl MON 0
1700ms  ctrl MON 1
1720ms  ctrl MON 0
2s      end
